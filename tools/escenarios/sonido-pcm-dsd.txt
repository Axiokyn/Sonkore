# PCM a DSD en la DX1 de tests/asound (ALSA nulo: tests/pipewire/alsa-null.conf),
# con el mouse en Ajustes › Audio › DSD mientras suena 04-tono-96k.flac (1 kHz
# a -12 dBFS): el modulador lleva el fondo de escala PCM al 45 % (-6,9 dB),
# así que el tono sale a -18,9 dB en DSD. Nativo a DSD64, DoP a DSD128 y
# apagado (vuelve a PCM a 96 kHz). `sonido` mide el tono desde la densidad de
# unos de cada palabra (dsd_measures en tools/ui-live.py).
pantalla 2546 4200
ajuste widgets/windows/settings=@Size(920 2700)
ajuste appearance/language=es
preparar python3 {repo}/tools/tonos.py --biblioteca {rt}/p
ajuste library/rescanOnStartup=true
ajuste playback/repeat=2
ajuste playback/device="alsa|hw:CARD=DX1,DEV=0"
argumentos {repo}/target/ui-live/tonos/04-tono-96k.flac
estado playing=true 5
sonido dsd=no tasa=96000 frec-i=1000±1 nivel-i=-12±0.3
tecla ctrl+comma
ventanas 2
foco last
esperar 0.8
clic 578 67
esperar 0.8

estado playing=true
